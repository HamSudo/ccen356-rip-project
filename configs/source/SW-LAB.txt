! ============================================================
! SW-LAB - Laboratory switch (Cisco Catalyst 3560-8PC)
! CCEN 356 Project 1 - RIPv2 school network, all-Ethernet design
!
! Ports:
!   Fa0/7        trunk to LAB-R3 G0/0 (VLAN 30 LAN + VLAN 23 WAN)
!   Fa0/8        access VLAN 23, to TCH-R2 G0/1 (the TCH-LAB WAN link)
!   Fa0/1 - 6    access VLAN 30: PC-LAB-1, PC-LAB-2, PC-LAB-3, Printer-LAB,
!                AUTO-PC (Fa0/5, runs the Python scripts)
!
! NOTE: this is a paste-in script (what gets typed into a blank device),
! not an export from the built network. The brief's ".txt exports from
! routers" are the 'show running-config' output, saved after the build
! in configs/exports/. Lines starting with '!' are comments, so the whole
! file can still be pasted into the CLI.
! ============================================================
enable
configure terminal
!
hostname SW-LAB
no ip domain-lookup
service password-encryption
enable secret class123
banner motd # SW-LAB - Laboratory switch. Authorized access only. #
!
! ---------- VLANs ----------
vlan 30
 name LAB-LAN
 exit
vlan 23
 name WAN-TCH-LAB
 exit
!
! ---------- Router ports ----------
! portfast on every router port: without it a port that comes back up
! (T3 restore) waits 30 s in spanning-tree before forwarding, which
! would be counted as RIP convergence time.
interface FastEthernet0/7
 description Trunk to LAB-R3 G0/0
 switchport trunk encapsulation dot1q
 switchport trunk allowed vlan 23,30
 switchport mode trunk
 switchport nonegotiate
 spanning-tree portfast trunk
 exit
interface FastEthernet0/8
 description WAN VLAN 23 - to TCH-R2 G0/1
 switchport mode access
 switchport access vlan 23
 spanning-tree portfast
 exit
!
! ---------- End devices ----------
interface range FastEthernet0/1 - 6
 description Lab end devices and automation PC
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 exit
!
! ---------- Management IP (in the LAN VLAN) ----------
interface Vlan30
 ip address 192.168.30.2 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.30.1
!
! ---------- Remote login ----------
! Telnet, not SSH: the switches run the C3560-IPBASE image, which has no
! crypto, so "crypto key" and "ip ssh" are rejected. The Python scripts only
! SSH to the routers; the switches are managed from the console.
ip domain-name school.local
username admin privilege 15 secret Admin@123
!
line console 0
 password cisco
 login
 logging synchronous
 exit
line vty 0 15
 login local
 transport input telnet
 exit
!
end
write memory
